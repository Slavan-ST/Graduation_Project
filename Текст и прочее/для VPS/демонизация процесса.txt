Для этого создадим файл в директории /etc/systemd/system/ с расширением .service


путь к директории приложения
/home/slavan/API/Основное приложение/ClientServer/WebAPI/bin/Release/net8.0/publish

путь к опубликованной dll
/home/slavan/API/Основное приложение/ClientServer/WebAPI/bin/Release/net8.0/publish/WebAPI.dll          


cd /etc/systemd/system/
sudo nano /etc/systemd/system/kestrel-test.service


!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!
[Unit]

Description=Example .NET Web API Application running on Ubuntu

[Service]
WorkingDirectory=/home/slavan/API/main/ClientServer/WebAPI/bin/Release/net8.0/publish
ExecStart=/home/slavan//API/main/ClientServer/WebAPI/bin/Release/net8.0/publish/WebAPI.dll
Restart=always
RestartSec=10 # Перезапускать сервис через 10 секунд при краше приложения
SyslogIdentifier=dotnet-example
User= slavan # пользователь, под которым следует запускать ваш сервис
Environment=ASPNETCORE_ENVIRONMENT=Production

[Install]
WantedBy=multi-user.target

!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!!



сам запуск сервисов

sudo systemctl enable kestrel-test.service
sudo systemctl start kestrel-test.service

проверка состояния 
sudo systemctl status kestrel-test.service
